
/* ---------- Solicitudes del formulario web (drycochile.cl → app + correo en español) ---------- */
var SOL = { PARA: 'contacto@drycochile.cl', APP: 'https://app.drycochile.cl', WA: '56956069678' };
function solTxt_(v, n) { return String(v == null ? '' : v).replace(/[\u0000-\u001f]/g, ' ').trim().slice(0, n || 200); }
function solEsc_(s) { return String(s == null ? '' : s).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); }
function solSlug_(s) { return String(s).toLowerCase().normalize('NFD').replace(/[̀-ͯ]/g, '').replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '').slice(0, 40) || 'cliente'; }
function solFono_(t) { var d = String(t || '').replace(/\D/g, ''); if (d.length === 9) d = '56' + d; if (d.length === 8) d = '569' + d; return d; }
function solFonoFmt_(d) { return d.length === 11 ? '+' + d.slice(0, 2) + ' ' + d.slice(2, 3) + ' ' + d.slice(3, 7) + ' ' + d.slice(7) : d; }

function solicitud(json) {
  var f = JSON.parse(json || '{}');
  if (f.web) return true;                                   // trampa para robots: se ignora en silencio
  var s = {
    nombre: solTxt_(f.nombre, 80), telefono: solTxt_(f.telefono, 30), email: solTxt_(f.email, 120).toLowerCase(),
    tipoCliente: solTxt_(f.tipoCliente, 40), servicios: (f.servicios || []).slice(0, 8).map(function (x) { return solTxt_(x, 60); }),
    comuna: solTxt_(f.comuna, 60), m2: solTxt_(f.m2, 20), urgencia: solTxt_(f.urgencia, 60),
    mensaje: solTxt_(f.mensaje, 2000), autoriza: !!f.autoriza, origen: solTxt_(f.origen, 120) || 'drycochile.cl'
  };
  var fono = solFono_(s.telefono);
  if (!s.nombre || fono.length < 8 || !s.comuna) throw new Error('FALTAN_DATOS');
  if (!s.mensaje) s.mensaje = '(sin descripción)';
  if (s.email && !/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(s.email)) throw new Error('EMAIL');
  var cache = CacheService.getScriptCache(), ck = 'sol_' + fono;
  if (cache.get(ck)) return true;                           // mismo teléfono hace menos de 2 minutos: no duplicar
  cache.put(ck, '1', 120);

  var now = new Date(), tz = 'America/Santiago', fecha = Utilities.formatDate(now, tz, 'yyyy-MM-dd'), hora = Utilities.formatDate(now, tz, 'HH:mm');
  var lock = LockService.getScriptLock(); lock.waitLock(20000);
  var cliId, nuevo = false, solId = 'sol-' + Utilities.formatDate(now, tz, 'yyMMddHHmmss') + '-' + Math.floor(Math.random() * 900 + 100);
  try {
    var sh = sheet_(), v = sh.getDataRange().getValues(), cli = null, row = -1;
    for (var i = 1; i < v.length; i++) {
      if (String(v[i][0]) !== 'clientes') continue;
      try {
        var o = JSON.parse(v[i][2]);
        if ((fono && solFono_(o.telefono) === fono) || (s.email && String(o.email || '').toLowerCase() === s.email)) { cli = o; cliId = String(v[i][1]); row = i + 1; break; }
      } catch (e) {}
    }
    var tipoMap = { 'Comunidad o edificio': 'Comunidad / edificio', 'Empresa o industria': 'Empresa', 'Particular': 'Persona' };
    var nota = 'Solicitud web ' + Utilities.formatDate(now, tz, 'dd-MM-yyyy') + ': ' + (s.servicios.join(', ') || 'sin servicio indicado') + '.';
    if (cli) {
      if (!cli.email && s.email) cli.email = s.email;
      if (!cli.telefono) cli.telefono = solFonoFmt_(fono);
      if (!cli.comuna) cli.comuna = s.comuna;
      cli.notas = (cli.notas ? cli.notas + '\n' : '') + nota;
      sh.getRange(row, 1, 1, 4).setValues([['clientes', cliId, JSON.stringify(cli), now]]);
    } else {
      nuevo = true; cliId = solSlug_(s.nombre) + '-' + Math.floor(Math.random() * 9000 + 1000);
      cli = { nombre: s.nombre, tipo: tipoMap[s.tipoCliente] || 'Persona', telefono: solFonoFmt_(fono), email: s.email || null, comuna: s.comuna,
              direccion: null, contacto: null, rut: null, reach: s.autoriza ? 'Sí' : 'No', notas: nota, origen: 'Formulario web', creado: now.getTime() };
      var r1 = sh.getLastRow() + 1; sh.getRange(r1, 1, 1, 4).setNumberFormats([['@', '@', '@', 'dd-mm-yyyy hh:mm']]).setValues([['clientes', cliId, JSON.stringify(cli), now]]);
    }
    var sol = { fecha: fecha, hora: hora, clienteId: cliId, clienteNuevo: nuevo, nombre: s.nombre, telefono: solFonoFmt_(fono), email: s.email || null,
                tipoCliente: s.tipoCliente || null, servicios: s.servicios, comuna: s.comuna, m2: s.m2 || null, urgencia: s.urgencia || null,
                mensaje: s.mensaje, autoriza: s.autoriza, estado: 'Nueva', origen: s.origen, creado: now.getTime() };
    var r2 = sh.getLastRow() + 1; sh.getRange(r2, 1, 1, 4).setNumberFormats([['@', '@', '@', 'dd-mm-yyyy hh:mm']]).setValues([['solicitudes', solId, JSON.stringify(sol), now]]);
  } finally { lock.releaseLock(); }

  try { solAviso_(s, fono, nuevo, now); } catch (e) { console.error('aviso', e); }
  if (s.email) { try { solConfirma_(s); } catch (e) { console.error('confirma', e); } }
  return true;
}

function solAviso_(s, fono, nuevo, now) {
  var urg = /filtraci/i.test(s.urgencia), pila = s.nombre.split(/\s+/)[0];
  var waTxt = 'Hola ' + pila + ', le saluda Nicolás Vergara de Dryco Chile. Recibimos su solicitud de cotización para ' + (s.servicios.join(', ').toLowerCase() || 'impermeabilización') + ' en ' + s.comuna + '. ¿Le acomoda coordinar una visita técnica?';
  var wa = 'https://wa.me/' + fono + '?text=' + encodeURIComponent(waTxt);
  var fila = function (k, val) { return val ? '<tr><td style="padding:9px 14px;color:#5b6b82;font-size:13px;width:38%;border-bottom:1px solid #e6ebf2">' + k + '</td><td style="padding:9px 14px;font-size:14px;color:#0b1f3a;font-weight:600;border-bottom:1px solid #e6ebf2">' + val + '</td></tr>' : ''; };
  var btn = function (href, txt, bg) { return '<a href="' + solEsc_(href) + '" style="display:inline-block;margin:4px 6px 4px 0;padding:11px 18px;border-radius:10px;background:' + bg + ';color:#fff;text-decoration:none;font-weight:700;font-size:14px">' + txt + '</a>'; };
  var html = '<div style="font-family:Arial,Helvetica,sans-serif;max-width:620px;margin:0 auto;background:#fff;border:1px solid #e6ebf2;border-radius:14px;overflow:hidden">'
    + '<div style="background:#0b1f3a;padding:18px 22px;color:#fff"><div style="font-size:12px;letter-spacing:2px;color:#5fd3f3;font-weight:700">NUEVA SOLICITUD DE COTIZACIÓN</div>'
    + '<div style="font-size:21px;font-weight:700;margin-top:6px">' + solEsc_(s.nombre) + '</div><div style="font-size:13px;color:#a9c4e8;margin-top:3px">' + solEsc_(s.comuna) + ' · ' + Utilities.formatDate(now, 'America/Santiago', "dd-MM-yyyy 'a las' HH:mm") + (nuevo ? ' · cliente nuevo' : ' · ya estaba en tu base de clientes') + '</div></div>'
    + (urg ? '<div style="background:#fdecec;color:#a61b1b;padding:10px 22px;font-weight:700;font-size:14px">Urgente: tiene una filtración activa</div>' : '')
    + '<div style="padding:16px 22px 6px">' + btn(wa, 'Responder por WhatsApp', '#25a244') + btn('tel:+' + fono, 'Llamar', '#1d5fd1') + (s.email ? btn('mailto:' + s.email, 'Responder por correo', '#0b1f3a') : '') + '</div>'
    + '<table style="width:100%;border-collapse:collapse;margin:8px 0">'
    + fila('Nombre', solEsc_(s.nombre)) + fila('Teléfono / WhatsApp', solEsc_(solFonoFmt_(fono))) + fila('Correo', solEsc_(s.email)) + fila('Tipo de cliente', solEsc_(s.tipoCliente))
    + fila('Servicio', solEsc_(s.servicios.join(', '))) + fila('Comuna de la obra', solEsc_(s.comuna)) + fila('Superficie aproximada', s.m2 ? solEsc_(s.m2) + ' m²' : '')
    + fila('¿Cuándo lo necesita?', solEsc_(s.urgencia)) + fila('Acepta recibir información', s.autoriza ? 'Sí' : 'No') + '</table>'
    + '<div style="padding:4px 22px 16px"><div style="font-size:13px;color:#5b6b82;margin-bottom:6px">Descripción del problema</div><div style="background:#f4f7fb;border-radius:10px;padding:12px 14px;font-size:14px;color:#0b1f3a;line-height:1.5;white-space:pre-wrap">' + solEsc_(s.mensaje) + '</div></div>'
    + '<div style="padding:12px 22px 20px;border-top:1px solid #e6ebf2;font-size:13px;color:#5b6b82">El cliente ya quedó registrado en la app. ' + btn(SOL.APP + '/#solicitudes', 'Abrir en la app', '#3b4cca') + '</div></div>';
  var plano = 'Nueva solicitud de cotización\n\nNombre: ' + s.nombre + '\nTeléfono: ' + solFonoFmt_(fono) + '\nCorreo: ' + (s.email || '-') + '\nTipo de cliente: ' + (s.tipoCliente || '-')
    + '\nServicio: ' + (s.servicios.join(', ') || '-') + '\nComuna: ' + s.comuna + '\nSuperficie: ' + (s.m2 || '-') + '\nCuándo: ' + (s.urgencia || '-') + '\n\n' + s.mensaje + '\n\nWhatsApp: ' + wa;
  var opt = { to: SOL.PARA, subject: (urg ? 'URGENTE · ' : '') + 'Nueva solicitud de cotización · ' + s.nombre + ' · ' + s.comuna, htmlBody: html, body: plano, name: 'Dryco Chile · Formulario web' };
  if (s.email) opt.replyTo = s.email;
  MailApp.sendEmail(opt);
}

    + '<div style="padding:22px 24px;color:#0b1f3a;font-size:15px;line-height:1.6">Hola ' + solEsc_(pila) + ',<br><br>Recibimos su solicitud de visita técnica y cotización'
    + (s.servicios.length ? ' para <b>' + solEsc_(s.servicios.join(', ').toLowerCase()) + '</b>' : '') + ' en <b>' + solEsc_(s.comuna) + '</b>.<br><br>'
    + 'Un especialista revisará su caso y lo contactará dentro de las próximas 24 horas hábiles para coordinar la visita.<br><br>'
    + 'Si tiene una filtración activa o prefiere hablar ahora, escríbanos por WhatsApp:<br>'

  MailApp.sendEmail({ to: s.email, subject: 'Recibimos su solicitud · Dryco Chile', htmlBody: html, name: 'Dryco Chile', replyTo: SOL.PARA,
    body: 'Hola ' + pila + ',\n\nRecibimos su solicitud de visita técnica y cotización en ' + s.comuna + '. Lo contactaremos dentro de las próximas 24 horas hábiles.\n\nWhatsApp: +56 9 5606 9678\n\nNicolás Vergara C.\nDryco Chile SpA' });
}
