function resumenDiario(forzar) {
  var t = hoy_(), dow = Utilities.formatDate(new Date(), AUTO.TZ, 'u');
  if (dow === '7' && forzar !== true) return; // domingo: sin correo
  var D = datos_(), byId = {};
  ['clientes', 'cotizaciones', 'obras'].forEach(function (c) { (D[c] || []).forEach(function (o) { byId[c + ':' + o.id] = o; }); });
  function cliO(id) { return byId['clientes:' + id] || null; }
  function cli(id) { var c = cliO(id); return c ? c.nombre : ''; }
  function conCli(n, id) { var c = cli(id); return c && String(n).indexOf(c) < 0 ? n + ' · ' + c : n; }
  function dias(a, b) { var x = a.split('-'), y = b.split('-'); return Math.round((new Date(+y[0], +y[1] - 1, +y[2]) - new Date(+x[0], +x[1] - 1, +x[2])) / 864e5); }
  function meses(a, b) { var x = a.split('-'), y = b.split('-'); return (+y[0] - +x[0]) * 12 + (+y[1] - +x[1]) - (+y[2] < +x[2] ? 1 : 0); }
  function waNum(id) { var c = cliO(id), n = c && c.telefono ? String(c.telefono).replace(/\D/g, '') : ''; if (n.length === 9) n = '56' + n; return n; }
  function saludo(id) {
    var c = cliO(id); if (!c) return 'Hola';
    if (!c.contacto && ((c.tipo && c.tipo !== 'Persona') || /^(edificio|comunidad|condominio|inmobiliaria|constructora|empresa|sociedad|club|colegio|iglesia|municipalidad|hotel|clínica|centro)\b/i.test(String(c.nombre || '').trim()))) return 'Hola';
    var w = String(c.contacto || c.nombre || '').split('·')[0].trim().split(/\s+/)[0]; return 'Hola' + (w ? ' ' + w : '');
  }
  function wa(id, txt) { return 'https://wa.me/' + waNum(id) + '?text=' + encodeURIComponent(txt); }
  function link(u, txt) { return '<a href="' + esc_(u) + '" style="color:#1F5CA1;font-weight:600;text-decoration:none">' + txt + '</a>'; }
  function maps(q) { return 'https://www.google.com/maps/search/?api=1&query=' + encodeURIComponent(q); }

  /* seguimiento de cotizaciones (mismas reglas que la app: 3, 7 y 14 días) */
  var SD = [3, 7, 14], SN = ['1er seguimiento', '2° seguimiento', 'último seguimiento'];
  var seg = [];
  (D.cotizaciones || []).forEach(function (c) {
    if (c.estado !== 'Enviada' || (c.segPausa && c.segPausa > t)) return;
    var n = Math.min(3, +c.segN || 0), log = c.segLog || [], ult = log.length ? log[log.length - 1].f : '';
    if (n >= 3) { if (dias(ult || c.fecha || t, t) >= 7) seg.push({ c: c, n: n, d: ult }); return; }
    var due = c.seguimiento || sumaDias_(c.fecha || t, SD[n]); if (c.segPausa && c.segPausa > due) due = c.segPausa;
    if (due <= t) seg.push({ c: c, n: n, d: due });
  });
  seg.sort(function (a, b) { return a.d.localeCompare(b.d); });
  function segText(c, n) {
    var num = c.numero + (c.fecha ? '/' + String(c.fecha).slice(0, 4) : ''), s = saludo(c.clienteId), sv = String(c.servicio || '').toLowerCase();
    if (n === 0) return s + ', le escribe Nicolás Vergara de Dryco Chile. Quería saber si pudo revisar la cotización N° ' + num + (sv ? ' de ' + sv : '') + ' que le enviamos. Si tiene dudas sobre la solución, los plazos o la garantía, con gusto se las aclaro.';
    if (n === 1) return s + ', ¿cómo está? Retomo la cotización N° ' + num + '. Si le acomoda, coordinamos una llamada corta para resolver dudas, o ajustamos la propuesta (alcance, etapas o fecha de inicio) según lo que necesite.';
    return s + ', le escribo por última vez sobre la cotización N° ' + num + ' para no molestarle. Si más adelante decide avanzar, la propuesta sigue disponible y puede escribirme directamente por aquí. ¡Que tenga un buen día!';
  }

  /* por cobrar */
  var obIx = {}; (D.obras || []).forEach(function (o) { obIx[o.id] = o; });
  var cob = {}; (D.cobros || []).forEach(function (x) { var o = obIx[x.obraId] || {}; var ci = x.conIva ? (x.conIva === 'Con IVA' || x.conIva === 'Sí') : o.conIva !== 'No'; cob[x.obraId] = (cob[x.obraId] || 0) + (+x.monto || 0) / (ci ? 1.19 : 1); });
  var porCobrar = (D.obras || []).filter(function (o) { return o.estado === 'Terminada'; }).map(function (o) {
    var f = o.conIva === 'No' ? 1 : 1.19, pn = (+o.neto || 0) - (cob[o.id] || 0);
    return { o: o, falta: pn > 5 ? Math.round(pn * f) : 0 };
  }).filter(function (x) { return x.falta > 0; });

  /* agenda de hoy con dirección y teléfono */
  var ag = (D.agenda || []).filter(function (a) { return !a.hecho && a.fecha === t; }).sort(function (a, b) { return (a.hora || '99').localeCompare(b.hora || '99'); });
  var man = (D.agenda || []).filter(function (a) { return !a.hecho && a.fecha === sumaDias_(t, 1); }).sort(function (a, b) { return (a.hora || '99').localeCompare(b.hora || '99'); });
  function agFila(a) {
    var o = a.obraId ? obIx[a.obraId] : null, c = cliO(a.clienteId) || (o ? cliO(o.clienteId) : null);
    var dir = a.direccion || (o && o.direccion ? [o.direccion, o.comuna].filter(Boolean).join(', ') : (c && c.direccion ? [c.direccion, c.comuna].filter(Boolean).join(', ') : ''));
    var extra = [];
    if (dir) extra.push(link(maps(dir), esc_(dir)));
    if (c && c.telefono) extra.push(link('tel:' + String(c.telefono).replace(/[^\d+]/g, ''), esc_(c.telefono)) + ' · ' + link('https://wa.me/' + waNum(c.id), 'WhatsApp'));
    return ['<b>' + (a.hora ? esc_(a.hora) + ' · ' : '') + esc_(a.titulo || a.tipo || '') + '</b><br><span style="color:#5B6B80">' + esc_([a.tipo, c && c.nombre].filter(Boolean).join(' · ')) + '</span>' + (extra.length ? '<br>' + extra.join(' · ') : ''), ''];
  }

  /* solicitudes sin contestar */
  var sol = (D.solicitudes || []).filter(function (s) { return !s.estado || s.estado === 'Nueva'; }).sort(function (a, b) { return ((b.fecha || '') + (b.hora || '')).localeCompare((a.fecha || '') + (a.hora || '')); });

  /* mantenciones por ofrecer */
  var mo = +t.slice(5, 7);
  var mant = (D.obras || []).filter(function (o) { return o.estado === 'Terminada' && o.termino && !o.mantNo; }).map(function (o) {
    var ms = meses(o.termino, t), last = o.mantContacto || '';
    if (ms >= 11 && (!last || meses(last, t) >= 10)) return { o: o, txt: 'cumple ' + Math.max(1, Math.round(ms / 12)) + (Math.round(ms / 12) > 1 ? ' años' : ' año') };
    if (mo >= 3 && mo <= 5 && ms >= 5 && (!last || last < t.slice(0, 4) + '-01-01')) return { o: o, txt: 'antes de las lluvias' };
    return null;
  }).filter(Boolean);

  var gar = (D.obras || []).filter(function (o) { return o.estado === 'Terminada' && (o.garantiaDesde || o.termino) && +o.garantia; }).map(function (o) {
    return { o: o, v: sumaMeses_(o.garantiaDesde || o.termino, o.garantia) };
  }).filter(function (x) { return x.v >= t && x.v <= sumaDias_(t, 30); });
  var activas = (D.obras || []).filter(function (o) { return o.estado === 'En curso' || (o.estado === 'Programada' && o.inicio === t); });

  function bloque(titulo, filas, vacio) {
    return '<h3 style="font:600 15px Arial;color:#0B1630;margin:22px 0 8px">' + titulo + '</h3>' +
      (filas.length ? '<table style="border-collapse:collapse;width:100%;font:14px Arial;color:#0D1B2E">' + filas.map(function (f) {
        return '<tr><td style="padding:8px 0;border-bottom:1px solid #E5EAF1;line-height:1.45">' + f[0] + '</td><td style="padding:8px 0 8px 10px;border-bottom:1px solid #E5EAF1;text-align:right;white-space:nowrap;vertical-align:top"><b>' + f[1] + '</b></td></tr>';
      }).join('') + '</table>' : '<p style="font:14px Arial;color:#5B6B80;margin:0">' + vacio + '</p>');
  }
  var totCob = porCobrar.reduce(function (a, x) { return a + x.falta; }, 0);
  var html = '<div style="max-width:620px;margin:auto;font:14px Arial;color:#0D1B2E">' +
    '<div style="background:#0B1630;color:#fff;padding:18px 20px;border-radius:12px"><div style="font:600 12px Arial;letter-spacing:2px;color:#8BE3DE">DRYCOCHILE APP</div>' +
    '<div style="font:700 20px Arial;margin-top:6px">Tu día · ' + fecha_(t) + '</div></div>' +
    bloque('Agenda de hoy (' + ag.length + ')', ag.map(agFila), 'Nada agendado para hoy.') +
    (man.length ? '<p style="font:13px Arial;color:#5B6B80;margin:10px 0 0"><b>Mañana:</b> ' + man.map(function (a) { return esc_((a.hora ? a.hora + ' ' : '') + (a.titulo || a.tipo || '')); }).join(' · ') + '</p>' : '') +
    (sol.length ? bloque('Solicitudes sin contestar (' + sol.length + ')', sol.slice(0, 10).map(function (s) {
      return [esc_(s.nombre || '') + ' · ' + esc_(s.canal || 'Web') + (s.comuna ? ' · ' + esc_(s.comuna) : '') + (s.telefono ? '<br>' + link('https://wa.me/' + String(s.telefono).replace(/\D/g, '').replace(/^9(\d{8})$/, '569$1'), 'Responder por WhatsApp') : ''), fecha_(s.fecha)];
    }), '') : '') +
    bloque('Cotizaciones por seguir (' + seg.length + ')', seg.slice(0, 12).map(function (x) {
      var c = x.c;
      return ['COT ' + esc_(c.numero) + ' · ' + esc_(cli(c.clienteId)) + '<br><span style="color:#5B6B80">' + (x.n >= 3 ? '3 seguimientos sin respuesta · ciérrala en la app' : SN[x.n] + ' · enviada hace ' + dias(c.fecha || t, t) + ' días') + '</span>' +
        (x.n < 3 ? '<br>' + link(wa(c.clienteId, segText(c, x.n)), 'Enviar mensaje por WhatsApp') : ''), clp_(c.neto)];
    }), 'Ninguna cotización por seguir hoy.') +
    (mant.length ? bloque('Mantenciones por ofrecer (' + mant.length + ')', mant.slice(0, 8).map(function (x) { return [esc_(conCli(x.o.nombre, x.o.clienteId)) + '<br><span style="color:#5B6B80">' + esc_(x.txt) + ' · ofrécela desde el Panel</span>', '']; }), '') : '') +
    bloque('Por cobrar · ' + clp_(totCob), porCobrar.map(function (x) { return [esc_(conCli(x.o.nombre, x.o.clienteId)), clp_(x.falta)]; }), 'Todas las obras terminadas están pagadas.') +
    (activas.length ? bloque('Obras en curso', activas.map(function (o) { return [esc_(o.nombre), (+o.avance || 0) + ' %']; }), '') : '') +
    (gar.length ? bloque('Garantías que vencen en 30 días', gar.map(function (x) { return [esc_(conCli(x.o.nombre, x.o.clienteId)) + ' · ofrece mantención', 'vence ' + fecha_(x.v)]; }), '') : '') +
    '<p style="margin:26px 0 0"><a href="https://app.drycochile.cl" style="background:#1F5CA1;color:#fff;text-decoration:none;padding:11px 18px;border-radius:10px;font:600 14px Arial">Abrir DrycoChile app</a></p>' +
    '<p style="font:12px Arial;color:#8A97A8;margin-top:22px">Correo automático de tu app. Llega de lunes a sábado a las 8:00.</p></div>';
  var asunto = ['DrycoChile · ' + fecha_(t)];
  asunto.push(ag.length ? ag.length + (ag.length > 1 ? ' actividades hoy' : ' actividad hoy') : 'sin agenda');
  if (sol.length) asunto.push(sol.length + (sol.length > 1 ? ' solicitudes nuevas' : ' solicitud nueva'));
  if (seg.length) asunto.push(seg.length + ' por seguir');
  MailApp.sendEmail({ to: AUTO.PARA, subject: asunto.join(' · '), htmlBody: html, name: 'DrycoChile app' });
}
